# ============================================================
# Stellaris Overmind — Autonomy Executor（游戏内自治执行层）
#
# 这是"手"。大模型写下的法案（overmind_lex.txt）是"法律"，
# 本文件是执法机关：每个月检查帝国状态，按法案日程表动手。
#
# 为什么必须放在游戏内：
#   Clausewitz 引擎不接受任何外部输入——它读不了磁盘文件，
#   也没有远程接口（已在 stellaris.exe 中确证）。游戏唯一愿意
#   读外部世界的时刻，是加载脚本的时候。所以外部大模型的决策
#   只能"编译进脚本"，之后由脚本在游戏内部持续执行。
#
# 八个相位（1-8），每月跑一个，六--八格轮转：
#   1 RELIEF     救济（补真正赤字的资源）
#   2 HOUSING    住房（城区）
#   3 JOBS       就业（乡村区划 + 专业区 zone）
#   4 AMENITIES  民生（舒适度建筑）
#   5 SCIENCE    科研与行政
#   6 MILITARY   军备（补船 + 政策）
#   7 ASCENSION  传统与飞升（树采纳 / 节点推进 / 飞升天赋）
#   8 WAR        战争与防御（轰炸姿态 / 陆军 / 星港）
#
# 费用模型（重要）：
#   add_district / add_zone / add_building / create_army /
#   create_starbase 都是**即时**生效、绕过建造队列的效果，游戏
#   不会收钱。所以每次建造都由本文件主动从国库扣费，扣多少钱
#   来自 overmind_charges.txt（由 engine/costs.py 从游戏自带数据
#   生成，即游戏自己的造价）。扣费前先用同源的 afford 触发器问
#   一次"付得起吗"——因为 add_resource 扣到 0 就夹住，不检查就
#   会凭空长出东西。传统同样按游戏 defines 的价格公式分档付费。
#
# 命名约定：所有新增键一律 overmind_ / om_ 开头（NFR-04）。
# 一切 ID、trigger、effect 名字都经过 engine/script_audit.py 对
# 游戏数据核验，名字写错会直接让测试失败，而不是静默失效。
# ============================================================


# ------------------------------------------------------------
# 通用建造助手（三件套：afford → charge → build）
#   参数：TYPE/ZONE/BUILDING = 游戏 ID
#         AFFORD/CHARGE     = 生成文件里的同源触发器和扣费效果
# ------------------------------------------------------------
# overmind_autonomy_do_district（已内联到各相位，避免多出一层嵌套 ——
# 引擎上限 5 层，拆成独立 effect 会顶到上限导致后续初始化被放弃）


# zone 不指定 zone_slot：官方脚本 151 次调用里有 69 次如此，
# 由引擎自己挑一个空槽位——比我们硬编码槽位索引安全得多。
# limit 里的 num_zones 保证同一颗行星不会重复建同一种专业区。
# overmind_autonomy_do_zone（已内联到各相位，避免多出一层嵌套 ——
# 引擎上限 5 层，拆成独立 effect 会顶到上限导致后续初始化被放弃）

	# D-41：入口清共旗 —— 自报只反映本相位动作（入口清共旗，非 per-phase 旗标）
	remove_country_flag = om_did_build

	# a) 法案指定的主要缺口资源，补一个对应的乡村区划
	if = {
		limit = {
			NOT = { has_country_flag = om_did_build }

				overmind_charge_district_district_mining = yes

			}
			add_district = {
				district_type = district_mining
			}
			log = "OVERMIND: 建了采矿区划（district_mining）于 [This.GetName]"
		}
	}

	# a2) S-19 救火（裁定 §2）：矿物库存 <200（收入打平也救）→ 建采矿区划，
	# 专用垫 100（破鸡生蛋；走统一花销门，能源收入门不豁免 P6）。
	# 这是对 D-28「库存判据盲区」的直接消费。

		limit = {
			NOT = { has_country_flag = om_did_build }
			overmind_autonomy_low_minerals = yes
			overmind_afford_district_district_mining_relief = yes
		}
		random_owned_planet = {
			limit = {
				num_free_districts = {

				overmind_charge_district_district_mining = yes

				district_type = district_mining
			}
			log = "OVERMIND: 建了采矿区划（district_mining）于 [This.GetName] —— 矿物库存救火"
		}
	}

	# b) 法案指定的专业区方向 → 一个 zone

			limit = {
				has_district = district_city
				num_zones = {
					type = zone_factory
					value = 0
				}
				owner = { overmind_afford_zone_zone_factory = yes }
			}
			owner = {
				overmind_charge_zone_zone_factory = yes
				set_country_flag = om_did_build
			}

				district = district_city
				zone = zone_factory
			}
			log = "OVERMIND: 【消费品】建了民用工厂区（zone_factory）—— 赤字优先"
		}
	}
	if = {
		limit = { NOT = { has_country_flag = om_did_build } }
		if = {
			limit = { check_variable = { which = om_lex_focus_zone value = 1 } }

					has_district = district_city

					zone = zone_industrial
				}
				log = "OVERMIND: 建了工业区（zone_industrial）于 [This.GetName]"
			}
		}
		else_if = {
			limit = { check_variable = { which = om_lex_focus_zone value = 2 } }
			random_owned_planet = {
				limit = {
					has_district = district_city
					num_zones = {

					has_district = district_city

				}

					zone = zone_research
				}
				log = "OVERMIND: 建了研究区（zone_research）于 [This.GetName]"
			}
		}
		else_if = {
			limit = { check_variable = { which = om_lex_focus_zone value = 4 } }

# 每月建舰（在 tick 层直接调用，**不经** run_slot / phase_military）
# ------------------------------------------------------------
# 放在这一层是为了保住引擎的 5 层嵌套预算：``create_military_fleet`` 自带
# 一个 ``effect`` 容器，如果还留在 phase_military 里，链会是
#     event.immediate(1) -> tick(2) -> run_slot(3) -> phase_military(4)
#         -> create_military_fleet(5) -> effect(6)   = 6，超限。
# 提到 tick 层后是 1-2-3-4，安全。详见 phase_military 顶部的说明。
#
# 用 ``create_military_fleet``（而非 ``create_fleet``）的理由：
#   1. 语义精准 —— 建的就是军用舰队，不会像 ``every_owned_fleet`` 那样
#      把军舰塞进民用编制（实机报「无法将民用舰船与其他舰船混合编制」）；
#   2. ``owner`` 参数在块外给定，effect 块里只需 ``create_ship``，
#      比 ``create_fleet`` 少写 set_owner / set_location。
# 用法取自 exe 自带的 effect 文档表：
#     create_military_fleet = { owner = <target> scaled_size = 1.0 effect = { } }
overmind_autonomy_build_ship = {

	# 建舰策略（2026-09-15 定稿，用户指令 + 玩家社区共识）：
	#   ① 发展优先 —— 前 10 游戏年零扩军，先把科研/殖民地/经济铺开
	#      （社区共识：前期合金是扩张命脉，舰队只求吓阻）；
	#   ② 资源冗余 —— 能源月收入必须为正（舰队维护吃能源，赤字扩军
	#      是社区公认的崩溃起点），合金安全垫 300（造完剩 200+）；
	#   ③ 威胁例外 —— 法案判威胁 ≥3 时豁免①②（先有船，再谈攒钱），
	#      合金门槛降回 50，但容量门与能源门不豁免；
	#   ④ 容量门（D-20）—— used_naval_capacity_percent < 1.0，超容
	#      维护惩罚重，缺省姿态不超容扩军（原版 ai_budget 同款触发器）。
	if = {
		limit = {
			NOT = { has_country_flag = om_did_ship }
			# S-2：容量门按年代查表（执行手册时代总表）——
			#   era2 巩固期 30-60%、era3 专业化期 60-90%、era4+ 满容量。
			#   era1 发展期无分支 → 不建舰；om_era 缺失（异常存档）时
			#   所有分支都不成立 → 同样不建舰，安全兜底。
			OR = {
				AND = {
					check_variable = { which = om_era value = 2 }
					used_naval_capacity_percent < 0.6
				}
				AND = {
					check_variable = { which = om_era value = 3 }
					used_naval_capacity_percent < 0.9
				}
				AND = {
					check_variable = { which = om_era value >= 4 }
					used_naval_capacity_percent < 1.0
				}
			}
			OR = {
				years_passed >= 10
				check_variable = { which = om_lex_threat value >= 3 }
				is_at_war = yes
			}
			OR = {
				# S-1：统一花销门 —— 合金安全垫 300（威胁 ≥3 时降到 50），
				# 能源收入为正的门在触发器内部，任何分支都不豁免。
				overmind_can_afford = { RESOURCE = alloys PAD = 300 }
				AND = {
					check_variable = { which = om_lex_threat value >= 3 }
					overmind_can_afford = { RESOURCE = alloys PAD = 50 }
				}
			}
		}
		create_military_fleet = {
			owner = root
			scaled_size = 1.0
			effect = {
				set_location = root.capital_scope
				create_ship = {
					name = random
					design = "NAME_AI_Core"
				}
			}
		}
		# 扣款与国家标记只做一次。原实现把它们写在 every_owned_fleet 内部，
		# 于是「每多一个舰队就多扣 100 合金」—— 一个安静的记账错误。
		set_country_flag = om_did_ship
		add_resource = { alloys = -100 }
		# R-2（三席评审红队#4）：om_did_ship 是月度标记，月初清零，
		# 而年度自动存档只会落在 slot 4/8 —— 军事相位建舰的证据几乎必然
		# 采样不到，fleet_size 于是成了唯一且无交叉验证的代理。
		# 这个累计计数器是第二证据源：探针读它即可独立确证"真的建过舰"。
		change_variable = { which = om_ships_built value = 1 }
		log = "OVERMIND: 新建了一支军用舰队（1 艘 NAME_AI_Core），合金 -100"
	}
}

			limit = {
				is_machine_empire = no
				is_natural_design_empire = no
				has_valid_civic = civic_augmentation_bazaars
			}
			set_variable = { which = om_ap_route value = 4 }
			add_ascension_perk = ap_the_flesh_is_weak
			log = "OVERMIND: 【飞升路线】 义体（Augmentation Bazaars 民议专厅）"
		}
		else_if = {
			limit = {
				is_machine_empire = no

			}
		}

			policy = economic_policy
			option = economic_policy_civilian
		}
	}

			overmind_autonomy_deficit_energy = yes

				option = economic_policy_civilian
			}

# 分量和上限 2+2+1+1 = 6，末尾夹紧回 5，保证 0..5 分级不破。
# 每个名字都在游戏语料里数过出现次数（D-8 方法），写错会被
# script_audit 的 unknown-key 检查直接拦下。

overmind_evaluate_threat = {
	set_variable = { which = om_lex_threat value = 0 }
	if = {
		limit = {
			any_neighbor_country = {
				relative_power = { who = root category = fleet value > equivalent }
			}
		}
		change_variable = { which = om_lex_threat value = 2 }
	}
	else_if = {
		limit = {
			any_neighbor_country = {
				relative_power = { who = root category = fleet value >= inferior }
			}
		}
		change_variable = { which = om_lex_threat value = 1 }
	}
	if = {
		limit = {
			any_war = {
				any_defender = { is_same_value = root }
			}
		}
		change_variable = { which = om_lex_threat value = 2 }
	}
	else_if = {
		limit = { is_at_war = yes }
		change_variable = { which = om_lex_threat value = 1 }
	}
	if = {
		limit = {
			any_neighbor_country = {
				OR = {
					is_rival = root
					has_valid_civic = civic_fanatic_purifiers
					has_valid_civic = civic_hive_devouring_swarm
					has_valid_civic = civic_machine_terminator
				}
			}
		}
		change_variable = { which = om_lex_threat value = 1 }
	}
	if = {
		limit = {
			OR = {
				has_global_flag = ai_invasion_happened
				has_global_flag = extradimensional_invasion_happened
				has_global_flag = prethoryn_invasion_happened
			}
		}
		change_variable = { which = om_lex_threat value = 1 }
	}
	if = {
		limit = { check_variable = { which = om_lex_threat value >= 5 } }
		set_variable = { which = om_lex_threat value = 5 }
	}
	log = "OVERMIND: 【威胁评估】 om_lex_threat 年度重算完成（邻国军力/战争/宿敌/危机四分量，0-5 分级）"
}

#   1. om_leader_count 从未 set_variable 初始化，而 v1 依赖 check_variable 的
#      当前值来判断「该不该补」—— 语义上要求「追踪真实领袖数」，却没有任何
#      代码把真实领袖数写进去（change_variable 只自增自己）= 自造账本，必然漂移。
#   2. assign_leader 段落是**空壳**（只有 log），日志谎报「已赋舰队」。
#   3. 阈值 3 是硬编码魔数，与「舰队无领袖」的真实触发条件无关。
#
# 【v2 的设计原则：不记账，实时查】—— 与 S-18 C2 的教训一致：
#   能用引擎的触发器直接问，就不要自己维护影子计数。
#
# 【本机取证】（D 盘游戏语料，非推断）
#   create_leader  : 116+82+52+40 处；参数 class/species/name/skill/traits/effect
#   assign_leader  : 语料有 last_created_leader（L1252/1336/1420，create_fleet 块内
#                    同层）与 event_target（L1018）两形态 —— ⚠️ 实机裁定（2026-09-18）：
#                    last_created_leader 跨 country→fleet **不贴上**（新局 13×重复创建），
#                    跨 scope 一律用 event_target 形态（L1003-1018 contingency 先例）
#     => 必须写在 **fleet scope** 内
#   set_leader     : 另一形态（L6702），本处不用
#   any_owned_fleet: 我方 autonomy.txt L2185/2192 已有使用先例
#
# 【层级风险规避】
#   本效果由 tick（第 4 层）直接调用 => 本效果体 = 第 5 层；
#   其内嵌的 if/limit/any_owned_fleet 均落在第 6 层，**不触及第 7 层上限**。
#   （对照：create_fleet 包一层就超限 —— 见本文件 §build_ship 先例注释）

overmind_autonomy_create_commander = {
	# 创建一个指挥官。skill = 3 对齐原生范例（00_scripted_effects.txt L1000）。
	# traits 用随机化，避免与原生 traits 列表耦合。
	# S-15b（2026-09-20 实机）：assign_leader = last_created_leader 跨 scope 不贴上
	# （新局连续 13 个月重复创建 = 舰队始终无领袖）—— 改原版标准形态：
	# 领袖自身 scope 里 save_event_target_as，舰队里按 event_target 赋任
	# （先例：00_scripted_effects.txt L1003-1018 contingency_fleet_leader）。
	create_leader = {
		class = commander
		name = random
		species = owner_main_species
		skill = 3
		randomize_traits = yes
		effect = {
			save_event_target_as = om_new_commander
		}
	}
	log = "OVERMIND: 【领袖】 创建指挥官（skill 3）—— 补充舰队指挥链"
}

overmind_autonomy_manage_leaders = {
	# S-15 v3（用户口径 2026-09-18）：三职业都招、空缺即补、总量守定额。
	#   指挥官 -> 无领袖「军用」舰队（S-15c 军用过滤保留）
	#   科学家 -> 无领袖「科研」舰队（shipclass_science_ship，语料 00:7605）
	#   官员/总督 -> 未实现（岗位检测复杂，S-15e 登记）
	# 上限门：引擎**未暴露容量触发器**（triggers.log 零命中，defines 只有 XP_SCALE）
	# ⇒ 用定额 count_owned_leader（先例 federations_election_events.txt:2397），
	#   定额 6 ≤ 基础容量下限 —— 宁少勿超（用户红线：招募领袖别超上限）。
	# 两个 hire 共享一道定额门；event_target 按职业分名，避免同 tick 相互覆盖。
	if = {
		limit = {
			count_owned_leader = {
				limit = { always = yes }
				count < 6
			}
		}
		if = {
			limit = {
				any_owned_fleet = {
					is_mobile = yes
					any_owned_ship = { is_ship_class = shipclass_military }   # S-15c：只补军用舰队
					NOT = { exists = leader }   # 舰队无领袖（实机验证：leader = {} 从 fleet
					# scope 直接切换是 Invalid context switch —— error.log 5 支舰队各 1 条，
					# 整个 if 恒假静默失效。exists = leader 是安全的存存在判定，
					# 舰队 scope 先例：nomads_effects.txt:792、caravaneer_scripted_effects.txt:212）
				}
			}
			overmind_autonomy_create_commander = yes
			# S-15b：必须写在 fleet scope 内（语料先例 L1018），且用 event_target
			# （last_created_leader 跨 country→fleet 实机不贴上，见 create_commander 注释）。
			random_owned_fleet = {
				limit = {
					is_mobile = yes
					any_owned_ship = { is_ship_class = shipclass_military }   # S-15c 同上
					NOT = { exists = leader }   # 同上：exists 判定，不做 scope 切换
				}
				assign_leader = event_target:om_new_commander
				log = "OVERMIND: 【领袖】 指挥官已就任该舰队（实时判定，无计数缓存）"
			}
		}
		if = {
			limit = {
				any_owned_fleet = {
					is_mobile = yes
					any_owned_ship = { is_ship_class = shipclass_science_ship }   # 科研船
					NOT = { exists = leader }
				}
			}
			overmind_autonomy_create_scientist = yes
			random_owned_fleet = {
				limit = {
					is_mobile = yes
					any_owned_ship = { is_ship_class = shipclass_science_ship }
					NOT = { exists = leader }
				}
				assign_leader = event_target:om_new_scientist
				log = "OVERMIND: 【领袖】 科学家已就任该科研舰队"
			}
		}
	}
}

overmind_autonomy_prune_leaders = {
	# S-15d（2026-09-18 用户实测：「招一堆指挥官把资源吃干了」）——
	# S-15b 历史污染的冗余指挥官（~200+ 名）留存在存档里持续扣维护（超容 → 维护倍增）。
	# 定编出清：无岗位指挥官（is_idle，引擎文档 triggers.log:1341）保留 ≤2 名 bench，
	# 超出的一次性解雇。三重豁免防误杀：带舰队/军队岗位者非 idle；内阁 is_councilor；
	# 元首 is_ruler。kill_leader 原版形态 gray_goo_effects.txt:946（show_notification = no
	# 不弹窗）。count_owned_leader / every_owned_leader 均引擎文档确证（4193/4989）。
	if = {
		limit = {
			count_owned_leader = {
				count > 2
				limit = {
					leader_class = commander
					is_councilor = no
					is_ruler = no
					is_idle = yes
				}
			}
		}
		every_owned_leader = {
			limit = {
				leader_class = commander
				is_councilor = no
				is_ruler = no
				is_idle = yes
			}
			kill_leader = { show_notification = no }
		}
		log = "OVERMIND: 【领袖】 超编出清 —— 解雇全部无岗位指挥官（定编 bench 2）"
	}
}

	# S-15e 实验（WDMSG-078）：官员自动入阁假说验证（一次性创建 + 月度自报）。
	# 判读：is_councilor=yes 的官员出现 ⇒ 引擎自动补位；恒无 ⇒ 硬边界候选。
	# count_owned_leader 先例：federations_election_events.txt:2397；is_councilor: triggers.log:2927。
overmind_autonomy_s15e_experiment = {
	if = {
		limit = { NOT = { has_country_flag = om_s15e_created } }
		set_country_flag = om_s15e_created
		create_leader = {
			class = official
			name = random
			species = owner_main_species
			skill = 2
			randomize_traits = yes
		}
		log = "OVERMIND: 【实验】S-15e 官员已创建 —— 观察引擎是否自动补内阁位"
	}
	if = {
		limit = {
			count_owned_leader = {
				count >= 1
				limit = {
					leader_class = official
					is_councilor = yes
				}
			}
		}
		log = "OVERMIND: 【实验】内阁中有官员在任（引擎补位成功）"
	}
	else = {
		log = "OVERMIND: 【实验】内阁中无官员（等待引擎补位或判硬边界）"
	}
}
