#   overmind_afford_<kind>_<id>  触发器：国库够不够付

# 这样帝国永远不可能在建不起的时候凭空长出东西。
#
# ⚠️ 这一对**分居两个目录**，不要把它们合并回去：
#   trigger 只有放在 common/scripted_triggers/ 才会被注册；
#   放进 scripted_effects/ 的顶层 trigger 会被静默丢弃，
#   运行时表现为 `Error in scripted trigger, cannot find: ...`。
#
# 写成字面数字而不是 @ 脚本变量：@ 在 $参数$ 展开之后的解析
# 顺序无法静态验证，而扣费出错是静默的。
# ============================================================

# ============================================================
# 传统的真实价格（由 defines 公式推导，按已采纳传统树分档）
#   cost(n) = base + (linear * n) ** exponent
#   n 取 num_tradition_categories * {per_tree}（游戏未暴露传统计数）
#   base=300 linear=8.0 exponent=1.8
# ============================================================
overmind_afford_tradition = {
	if = {
		limit = { num_tradition_categories < 1 }
		resource_stockpile_compare = { resource = unity value >= 300 }
	}
	else_if = {
		limit = { num_tradition_categories < 2 }
		resource_stockpile_compare = { resource = unity value >= 1702 }
	}
	else_if = {
		limit = { num_tradition_categories < 3 }
		resource_stockpile_compare = { resource = unity value >= 5182 }
	}
	else_if = {
		limit = { num_tradition_categories < 4 }
		resource_stockpile_compare = { resource = unity value >= 10429 }
	}
	else_if = {
		limit = { num_tradition_categories < 5 }
		resource_stockpile_compare = { resource = unity value >= 17300 }
	}
	else_if = {
		limit = { num_tradition_categories < 6 }
		resource_stockpile_compare = { resource = unity value >= 25703 }
	}
	else_if = {
		limit = { num_tradition_categories < 7 }
		resource_stockpile_compare = { resource = unity value >= 35570 }
	}
	else = {
		resource_stockpile_compare = { resource = unity value >= 35570 }
	}
}
