# 消费品月度净收入为负（2026-09-16 用户反馈「消费品一直赤字」新增）。
# 消费品是最容易被忽视的隐性赤字：研究员/职员/娱乐岗位都在消耗它，
# 而它不直接决定产能，只在破产（bankruptcy）时才让人察觉。

	has_monthly_income = {
		resource = consumer_goods
		value < 0
	}
}

# （实机 error.log: "Scripted Trigger overmind_autonomy_deficit_alloys is invalid"）。

	}

# S-1（策略规格书 §3）：统一花销门 —— 全部花销动作共用
#   ① 库存不得低于安全垫（$PAD$）；
#   ② 能源月收入必须为正（舰队/建筑维护都吃能源，赤字扩军
#      是社区公认的崩溃起点）。能源门在任何姿态下都不豁免。
# 用法：overmind_can_afford = { RESOURCE = alloys PAD = 300 }
# ------------------------------------------------------------
overmind_can_afford = {
	resource_stockpile_compare = { resource = $RESOURCE$ value >= $PAD$ }
	resource_income_compare = { resource = energy value > 0 }
}


# ------------------------------------------------------------
# S-3（策略规格书 §4.3）：宣战六条件评估器
#
# 评估与执行分离：本触发器只回答「这个邻国现在是不是合格目标」，
# 真正的 declare_war + 战争目标（wg_*）归 M3-4（依赖 M3-1 调研）。
# 调用方式：any_neighbor_country = { overmind_can_declare_war = yes }
# —— 在目标国 scope 内求值，root = 我方。
#
# 六条件对照（全部满足才宣战）：
#   ① years_passed >= 30（专业化期之后）；且 < 150 封顶 —— 手册 era5
#     「仅防御反击」，危机准备期不开新战端；
#   ② 目标军力明显弱于我方。⚠️ D-24：relative_power 的 value 只接受定性
#     token —— 原写法 `value < 0.7` 实机每月刷 22 条 error 且静默判假。
#     现用 `<= inferior`（inferior 档 ≈ ≤0.67x；Paradox Wiki Empire 页 +
#     defines 710-711 交叉确证），与规格书「< 70%」意图几乎精确对应。
#     方向：在目标 scope 内 who = root，value = 目标/我方。
#   ③ 我方能源/合金月收入为正 + 合金库存达安全垫 300（统一花销门）；
#   ④ 目标不在联邦（has_federation，vanilla 高频触发器）—— v1 取规格书「目标国不在联邦**或**其联邦不在
#     战争状态」的第一析支，联邦战状态检查留 M3-1 语料研究后放宽；
#   ⑤ 不打灭国战 —— 触发器层的近似：灭国型民议目标（清洗者/吞噬蜂群/
#     终结者）剔除，这类目标没有「有限战争」可言；战争目标本身的
#     有限性约束在 M3-4 落地；
#   ⑥ 威胁 >=3（S-5 动态评估值）豁免条件③的库存垫 —— 但能源收入门
#     任何分支不豁免（P6）。「对方已宣战」在和平期评估语境下不可能
#     成立（战争是双边的，彼时我方必已开战、走 WAR 相位战时分支），
#     故豁免只收敛为威胁一项。
# ------------------------------------------------------------
overmind_can_declare_war = {
	years_passed >= 30
	years_passed < 150
	relative_power = { who = root category = fleet value <= inferior }
	has_federation = no
	NOT = { has_valid_civic = civic_fanatic_purifiers }
	NOT = { has_valid_civic = civic_hive_devouring_swarm }
	NOT = { has_valid_civic = civic_machine_terminator }
	OR = {
		AND = {
			root = { resource_income_compare = { resource = energy value > 0 } }
			root = { resource_income_compare = { resource = alloys value > 0 } }
			root = { overmind_can_afford = { RESOURCE = alloys PAD = 300 } }
		}
		AND = {
			root = { check_variable = { which = om_lex_threat value >= 3 } }
			root = { resource_income_compare = { resource = energy value > 0 } }
			root = { resource_income_compare = { resource = alloys value > 0 } }
		}
	}
}
